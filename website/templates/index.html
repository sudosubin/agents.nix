Use these packages in your Nix configuration

Add the overlay, then refer to any package by its attribute path. Every package page shows the exact path to copy.

{
  inputs.agents-nix.url = "github:{{ repository }}";

  outputs = { nixpkgs, agents-nix, ... }:
    let
      pkgs = import nixpkgs {
        system = "aarch64-darwin";
        overlays = [ agents-nix.overlays.default ];
      };
    in
    {
      # pkgs.agent-skills.github.vercel-labs.skills.find-skills
    };
}

Or build one directly, without the overlay:

nix build github:{{ repository }}#agent-skills.aarch64-darwin.github.vercel-labs.skills.find-skills
